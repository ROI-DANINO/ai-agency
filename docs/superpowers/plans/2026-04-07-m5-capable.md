> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Human can manage a full feature from design through build entirely via `/org`. Multiple workflow types. Stop and resume working across sessions. Full session lifecycle (start → work → end → handoff).

**Architecture:** Add a workflow registry that maps workflow names to LangGraph graphs. Add `feature-design` and `feature-build` workflow manifests. Update `run-workflow.ts` to accept a workflow name and thread_id for resume. Update the `/org` skill to route into named workflows. Add `/org-end` skill for session close + handoff.

**Tech Stack:** YAML workflow manifests (new), existing LangGraph + Commander patterns, existing skill system.

**Prerequisites:** M1, M2, M3, M4 complete.

- [ ] **Step 3: Create `packages/cli/src/workflow/registry.ts`**

```typescript
import { readdir, readFile } from "fs/promises";
import { join } from "path";
import yaml from "js-yaml";

export interface WorkflowLeadDef {
  id: string;
  goal: string;
  tier: number;
  depends_on: string[];
  gate: "human" | "auto";
}

export interface WorkflowManifest {
  name: string;
  description: string;
  version: string;
  leads: WorkflowLeadDef[];
  timeout_minutes: number;
  max_retries: number;
}

export async function loadWorkflowRegistry(repoRoot: string): Promise<WorkflowManifest[]> {
  const dir = join(repoRoot, "config", "workflows");
  let files: string[];
  try {
    files = await readdir(dir);
  } catch {
    return [];
  }

  const manifests: WorkflowManifest[] = [];
  for (const file of files.filter(f => f.endsWith(".yaml"))) {
    const raw = await readFile(join(dir, file), "utf-8");
    const doc = yaml.load(raw) as Record<string, unknown>;

    const leads = (doc["leads"] as Array<Record<string, unknown>>).map(l => ({
      id: String(l["id"]),
      goal: String(l["goal"]),
      tier: Number(l["tier"] ?? 2),
      depends_on: (l["depends_on"] as string[]) ?? [],
      gate: String(l["gate"] ?? "auto") as "human" | "auto",
    }));

    manifests.push({
      name: String(doc["name"]),
      description: String(doc["description"]),
      version: String(doc["version"] ?? "1.0"),
      leads,
      timeout_minutes: Number(doc["timeout_minutes"] ?? 30),
      max_retries: Number(doc["max_retries"] ?? 2),
    });
  }

  return manifests;
}

export function interpolateGoals(
  manifest: WorkflowManifest,
  vars: Record<string, string>,
): WorkflowManifest {
  const leads = manifest.leads.map(lead => ({
    ...lead,
    goal: lead.goal.replace(/\{(\w+)\}/g, (_, key) => vars[key] ?? `{${key}}`),
  }));
  return { ...manifest, leads };
}

export function manifestToTaskYaml(
  manifest: WorkflowManifest,
  params: { task: string; threadId: string },
): string {
  const now = new Date().toISOString();
  const leads = manifest.leads.map(l => ({
    id: l.id,
    goal: l.goal,
    model: "auto",
    runtime: "auto",
    tier: l.tier,
    depends_on: l.depends_on,
    gate: l.gate,
    max_retries: null,
    retry_count: 0,
    status: "DRAFT",
    queued_at: null,
    started_at: null,
    completed_at: null,
    approved_at: null,
  }));

  return yaml.dump({
    task: params.task,
    thread_id: params.threadId,
    requested_by: "human",
    created_at: now,
    approved_at: null,
    completed_at: null,
    leads,
    timeout_minutes: manifest.timeout_minutes,
    max_retries: manifest.max_retries,
    mesh_check_interval: 5,
    hitl_log: [],
  });
}
```

```typescript
import { loadWorkflowRegistry, interpolateGoals, manifestToTaskYaml } from "../workflow/registry.js";
import { AsyncSqliteSaver } from "@langchain/langgraph-checkpoint-sqlite";

// If --resume is given, use the persisted thread_id
const threadId = opts.resume ?? randomUUID();

// If --workflow is given, load and interpolate the manifest
let taskToRun: string = opts.task ?? "No task specified";
if (opts.workflow) {
  const registry = await loadWorkflowRegistry(REPO_ROOT);
  const found = registry.find(w => w.name === opts.workflow);
  if (!found) {
    console.error(`Unknown workflow: ${opts.workflow}`);
    console.error(`Available: ${registry.map(w => w.name).join(", ")}`);
    process.exit(1);
  }

  // Parse --var flags: "feature=Auth system" → { feature: "Auth system" }
  const vars: Record<string, string> = {};
  for (const v of (opts.var ?? []) as string[]) {
    const eq = v.indexOf("=");
    if (eq > 0) vars[v.slice(0, eq)] = v.slice(eq + 1);
  }

  const interpolated = interpolateGoals(found, vars);
  taskToRun = manifestToTaskYaml(interpolated, { task: opts.task ?? found.description, threadId });
}

// Use SQLite checkpointer for persistence across restarts
const checkpointer = await AsyncSqliteSaver.fromConnString(
  join(REPO_ROOT, ".mesh", "workflow-state.db")
);
const { compiled, startWatcher } = buildWorkflowGraph(hermes, { checkpointer });
```

## Step 1 — Collect session work

Read these in order:

```markdown
## Artifacts Produced
{list artifacts with links to .mesh/docs/artifacts/}

{list decisions from .mesh/channels/decisions/}

{top 1-2 next actions based on current TASKS.md ## Now}
```
