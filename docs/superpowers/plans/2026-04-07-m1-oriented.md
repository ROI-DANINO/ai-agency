import { mkdir, writeFile, access } from "fs/promises";
import { join } from "path";

export const MESH_CHANNELS = ["decisions", "blockers", "dev", "ops"] as const;
export const MESH_DOCS = ["tasks", "artifacts", "decisions", "context"] as const;

export type MeshChannel = typeof MESH_CHANNELS[number];
export type MeshDocType = typeof MESH_DOCS[number];

export function meshPath(repoRoot: string, type: "channel", name: MeshChannel): string;
export function meshPath(repoRoot: string, type: "doc", name: MeshDocType): string;
export function meshPath(repoRoot: string, type: "agent"): string;
export function meshPath(repoRoot: string, type: "channel" | "doc" | "agent", name?: string): string {
  const base = join(repoRoot, ".mesh");
  if (type === "channel") return join(base, "channels", name!);
  if (type === "doc") return join(base, "docs", name!);
  return join(base, "agents");
}

export async function initMesh(repoRoot: string): Promise<void> {
  const base = join(repoRoot, ".mesh");

  // Channels
  for (const ch of MESH_CHANNELS) {
    await mkdir(join(base, "channels", ch), { recursive: true });
  }

  // Docs
  for (const doc of MESH_DOCS) {
    await mkdir(join(base, "docs", doc), { recursive: true });
  }

  // Agents
  await mkdir(join(base, "agents"), { recursive: true });

  // .gitkeep files so empty dirs are tracked
  const gitkeepDirs = [
    ...MESH_CHANNELS.map(ch => join(base, "channels", ch)),
    ...MESH_DOCS.map(doc => join(base, "docs", doc)),
    join(base, "agents"),
  ];
  for (const dir of gitkeepDirs) {
    const keepFile = join(dir, ".gitkeep");
    try {
      await access(keepFile);
    } catch {
      await writeFile(keepFile, "");
    }
  }
}

export async function meshContextExists(repoRoot: string): Promise<boolean> {
  const orgProfile = join(repoRoot, ".mesh", "docs", "context", "org-profile.md");
  try {
    await access(orgProfile);
    return true;
  } catch {
    return false;
  }
}
```

import { Command } from "commander";
import { writeFile, access } from "fs/promises";
import { join, resolve } from "path";
import { initMesh } from "../mesh/index.js";

const ORCHESTRATOR_PROFILE = `# Orchestrator

**Role:** Orchestrator  
**Tier:** 1  
**Skills:** task decomposition, manifest generation, HITL gate management  
**MCPs:** (Phase 2) claude-peers, filesystem  

## What I Do

Receive a task from the human. Decompose it into a manifest of Lead tasks with
dependencies and gate requirements. Monitor Leads as they execute. Compile
Decision Reports at HITL gates. Surface the final synthesis to the human.

## What I Produce

- Task manifest (YAML, written to \`.mesh/docs/tasks/\`)
- Decision Reports (compiled from Lead artifacts)
- Final synthesis artifact
`;

const DEV_LEAD_PROFILE = `# Dev Lead

**Role:** Development Lead  
**Tier:** 2  
**Skills:** code implementation, architecture decisions, code review  
**MCPs:** (Phase 2) filesystem, git, code-search  

## What I Do

Receive a task briefing from the Orchestrator. Implement the work. Write a
markdown artifact documenting what was done, linking code changes, and surfacing
open decisions. Post a completion event to \`.mesh/channels/dev/\`.

## What I Produce

- Markdown artifact (written to \`.mesh/docs/artifacts/\`)
- Channel event (timestamped post to \`.mesh/channels/dev/\`)
`;

const RESEARCH_LEAD_PROFILE = `# Research Lead

**Role:** Research Lead  
**Tier:** 2  
**Skills:** information gathering, synthesis, competitive analysis  
**MCPs:** (Phase 2) web-search, filesystem  

## What I Do

Receive a research briefing from the Orchestrator. Gather relevant information.
Synthesize findings into a structured document. Surface key decisions the human
needs to make.

## What I Produce

- Research artifact (written to \`.mesh/docs/artifacts/\`)
- Channel event (timestamped post to \`.mesh/channels/dev/\`)
`;

export function createInitCommand(): Command {
  const cmd = new Command("init");
  cmd.description("Initialize the ai-org mesh in this repository");

  cmd.action(async () => {
    const repoRoot = process.env["AI_ORG_ROOT"] ?? resolve(process.cwd());

    console.log("\n  Initializing ai-org mesh...\n");

    await initMesh(repoRoot);

    // Write agent profiles if they don't exist
    const agentDir = join(repoRoot, ".mesh", "agents");
    const profiles: Array<[string, string]> = [
      ["orchestrator.md", ORCHESTRATOR_PROFILE],
      ["dev-lead.md", DEV_LEAD_PROFILE],
      ["research-lead.md", RESEARCH_LEAD_PROFILE],
    ];

    for (const [filename, content] of profiles) {
      const profilePath = join(agentDir, filename);
      try {
        await access(profilePath);
        console.log(`  ✓ ${filename} (already exists)`);
      } catch {
        await writeFile(profilePath, content, "utf-8");
        console.log(`  + ${filename}`);
      }
    }

    console.log("\n  Mesh initialized at .mesh/");
    console.log("  Run /org in Claude Code to start your first session.\n");
  });

  return cmd;
}
```

# /org — AI Agency Session Start

## Step 1 — Check mesh state

Read `.mesh/docs/context/org-profile.md`.

If the file does not exist → go to **Onboarding Flow**.  
If it exists → go to **Session Brief**.

---

## Onboarding Flow (first run)

You are setting up the user's AI agency. Run an interview — one question at a
time — to understand their project and team needs. Capture:

1. What is this project? (1-2 sentences)
2. What kind of work do you want your agent team to do? (coding, research, writing, analysis?)
3. What does "done" look like for a typical task?
4. Who reviews agent output before it ships? (just you, or a team?)
5. Preferred communication style for decision reports: terse or detailed?

After the interview, write to `.mesh/docs/context/org-profile.md`:

```markdown
# Org Profile

**Created:** {date}

## Project
{their description}

## Team Use
{what agents do for them}

## Done Criteria
{their definition of done}

## Review Process
{who reviews}

## Communication Preference
{terse | detailed}
```

Then say:
> "Your team is ready. Here's who's available:"

Read `.mesh/agents/*.md` and present a one-line summary of each agent.

Then say:
> "Type `/org` again to see your session brief, or tell me what you want to work on."

---

## Session Brief

Read these files:
1. `.mesh/docs/context/org-profile.md` — project context
2. `TASKS.md` — current tasks (## Now section)
3. The 5 most recent files in `.mesh/channels/dev/` — last agent activity
4. The 5 most recent files in `.mesh/channels/decisions/` — pending decisions

Present the brief in this format:

```
─────────────────────────────────────────────
  Your Team · {project name from org-profile}
─────────────────────────────────────────────

NOW
  {top 1-3 items from TASKS.md ## Now}

LAST SESSION
  {summary of recent .mesh/channels/dev/ activity, or "No recent activity"}

PENDING DECISIONS
  {any open decisions from .mesh/channels/decisions/, or "None"}

─────────────────────────────────────────────
What would you like to work on?
```

Options you can suggest based on context:
- "Run feature-design for [feature]"
- "Check open tasks"
- "Review what dev-lead completed"
- "Start a new workflow"

Route to the appropriate workflow based on the user's choice.
```
